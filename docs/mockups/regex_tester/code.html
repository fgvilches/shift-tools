use regex::Regex;

fn parse_network_endpoints(text: &str) {
    // Compiled with zero-backtracking guarantee
    let re = Regex::new(r"(?P<protocol>https?)://(?P<host>[a-zA-Z0-9.-]+)(?::(?P<port>\d+))?(?P<path>/[^\s?#]*)?").unwrap();

    for caps in re.captures_iter(text) {
        println!("Protocol: {}, Host: {}, Port: {:?}, Path: {:?}",
            &caps["protocol"],
            &caps["host"],
            caps.name("port").map(|m| m.as_str()),
            caps.name("path").map(|m| m.as_str()),
        );
    }
}
