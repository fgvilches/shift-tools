// Rust: tokio-cron-scheduler
use tokio_cron_scheduler::{Job, JobScheduler};

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let sched = JobScheduler::new().await?;
    let job = Job::new_async("*/15 08-18 * * 1-5", |_uuid, _lock| {
        Box::pin(async move {
            println!("Executing periodic dispatch: {}", chrono::Utc::now());
        })
    })?;
    sched.add(job).await?;
    sched.start().await?;
    Ok(())
}
